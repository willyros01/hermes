#!/usr/bin/env bash
# FIDUNIO HELP 1.1.84: HELP alarm sound (functions only)
#
# ONE COMMAND, NO PARAMETERS:
#   bash hl3.txt
#
# This script:
# - explicitly switches to fidunio-fef13 and checks project number 130339622893
# - downloads the tested 1.1.83 HELP server code from GitHub and runs its test
# - writes a rollback script first (it puts back the HELP functions you
#   deployed with hlp.txt)
# - deploys ONLY the "help" functions (helpAlertCreatedV1, helpAlertUpdatedV1,
#   helpAlertSweepV1)
#
# What changes for people:
# - a HELP alert now asks the phone to play FIDUNIO's own alarm sound:
#   iPhone/iPad "help_alarm.caf", Android/Fire the "HELP alarm" channel
#   (needs FIDUNIO 1.1.84 on the helper's device; older versions keep the
#   ordinary sound). Ringing every 30 seconds until answered is unchanged.
# - it also re-checks the permission that lets the database call the HELP
#   functions (the hip.txt fix) and puts it back if a deploy removed it
#
# It does NOT change Firestore rules, Storage rules, IAM, App Check, any other
# function, the PIN, encryption keys, application data, TestFlight or the APK.
set -euo pipefail

PROJECT="fidunio-fef13"
PROJECT_NUMBER_EXPECTED="130339622893"
NEW_COMMIT="8d0ed4c3496a7a3f5c61ea1d43d63c19a4379b72"
OLD_COMMIT="910906b2b75e705bf390b8dc4dd968ea1496b450"
REPO="willyros01/hermes"

say(){ printf '\n==== %s ====\n' "$*"; }
stop(){ echo "STOP: $*" >&2; exit 1; }
for cmd in gcloud curl tar npm node; do command -v "$cmd" >/dev/null 2>&1 || stop "$cmd is required"; done

check_project(){
  gcloud config set project "$PROJECT" >/dev/null
  local active number
  active="$(gcloud config get-value project 2>/dev/null)"
  number="$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')"
  [[ "$active" == "$PROJECT" ]] || stop "Project switch failed; active project is '$active'."
  [[ "$number" == "$PROJECT_NUMBER_EXPECTED" ]] || stop "Wrong project number '$number'."
}

say "1/5 FIDUNIO project safety switch"
check_project
echo "PASS: target project is $PROJECT ($PROJECT_NUMBER_EXPECTED)."

STAMP="$(date -u +%Y%m%dT%H%M%SZ)"
BACKUP_DIR="$HOME/fidunio-help3-$STAMP"
WORK="$(mktemp -d)"
mkdir -p "$BACKUP_DIR"
trap 'rm -rf "$WORK"' EXIT

say "2/5 Download and test the 1.1.83 HELP server code"
curl --retry 3 -fsSL "https://github.com/$REPO/archive/$NEW_COMMIT.tar.gz" -o "$WORK/new.tgz"
tar -xzf "$WORK/new.tgz" -C "$WORK"
ROOT="$WORK/hermes-$NEW_COMMIT"
[[ -d "$ROOT/functions-help" ]] || stop "HELP code did not unpack."
grep -q "HELP_REALERT_MS=30\*1000" "$ROOT/functions-help/help-core.mjs" || stop "This is not the 30-second HELP code."
grep -q "help_alarm.caf" "$ROOT/functions-help/help-push.mjs" || stop "This is not the HELP alarm sound code."
( cd "$ROOT/functions-help" && npm ci --ignore-scripts --no-audit --no-fund >/dev/null && node help-core.test.mjs )

if ! command -v firebase >/dev/null 2>&1; then npm install -g firebase-tools; fi

write_fn_project(){ # $1 = source root, $2 = target dir
  mkdir -p "$2"
  cp -R "$1/functions-help" "$2/functions-help"
  cat > "$2/firebase.json" <<'JSON'
{
  "functions": [
    { "source": "functions-help", "codebase": "help", "ignore": ["node_modules", "*.test.mjs"] }
  ]
}
JSON
}

say "3/5 Rollback script (written before any change)"
curl --retry 3 -fsSL "https://github.com/$REPO/archive/$OLD_COMMIT.tar.gz" -o "$WORK/old.tgz"
tar -xzf "$WORK/old.tgz" -C "$WORK"
write_fn_project "$WORK/hermes-$OLD_COMMIT" "$BACKUP_DIR/old-help"
cat > "$BACKUP_DIR/rollback.sh" <<ROLLBACK
#!/usr/bin/env bash
# Puts back the HELP functions from hl2.txt (30-second ring, ordinary sound).
set -euo pipefail
gcloud config set project "$PROJECT" >/dev/null
[[ "\$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')" == "$PROJECT_NUMBER_EXPECTED" ]] || { echo "STOP: wrong project"; exit 1; }
cd "$BACKUP_DIR/old-help/functions-help" && npm ci --ignore-scripts --no-audit --no-fund >/dev/null
cd "$BACKUP_DIR/old-help" && firebase deploy --only functions:help --project "$PROJECT" --non-interactive
for svc in helpalertcreatedv1 helpalertupdatedv1; do gcloud run services add-iam-policy-binding "\$svc" --region=us-central1 --member="serviceAccount:fidunio-notification@$PROJECT.iam.gserviceaccount.com" --role=roles/run.invoker --format=none; done
echo "ROLLBACK COMPLETE: the previous HELP functions are back."
ROLLBACK
chmod +x "$BACKUP_DIR/rollback.sh"
echo "Rollback ready: $BACKUP_DIR/rollback.sh"

say "4/5 Deploy ONLY the HELP functions (codebase 'help')"
check_project
write_fn_project "$ROOT" "$WORK/fn"
( cd "$WORK/fn/functions-help" && npm ci --ignore-scripts --no-audit --no-fund >/dev/null )
( cd "$WORK/fn" && firebase deploy --only functions:help --project "$PROJECT" --non-interactive )

say "5/5 Check (and keep the hip.txt permission)"
for svc in helpalertcreatedv1 helpalertupdatedv1; do
  gcloud run services add-iam-policy-binding "$svc" --region=us-central1 \
    --member="serviceAccount:fidunio-notification@$PROJECT.iam.gserviceaccount.com" --role=roles/run.invoker --format=none
  echo "PASS: $svc may be called by the database (permission kept)"
done
LIST="$(gcloud functions list --regions=us-central1 --format='value(name)' 2>/dev/null || true)"
for f in helpAlertCreatedV1 helpAlertUpdatedV1 helpAlertSweepV1; do
  grep -q "$f" <<<"$LIST" || stop "$f is not listed after deploy. Rollback: bash $BACKUP_DIR/rollback.sh"
done
echo "PASS: helpAlertCreatedV1, helpAlertUpdatedV1, helpAlertSweepV1 are deployed."

say "SUCCESS"
echo "HELP alerts now ask for the FIDUNIO alarm sound (helpers need FIDUNIO 1.1.84)."
echo "HELP still rings every 30 seconds until someone answers."
echo "Backup and rollback: $BACKUP_DIR"
echo "Nothing else was changed."
