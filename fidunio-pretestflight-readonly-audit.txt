#!/usr/bin/env bash
# FIDUNIO one-time pre-TestFlight live Firebase audit.
# READ-ONLY: this script contains no deploy/create/update/delete/patch commands.
set -euo pipefail

PROJECT="fidunio-fef13"
PROJECT_NUMBER_EXPECTED="130339622893"
IOS_APP_ID="1:130339622893:ios:929f4e431c91ef7ed0e1e4"
WEB_APP_ID="1:130339622893:web:912a9016e294b30dd0e1e4"
REGION="us-central1"
OUT="$HOME/fidunio-readonly-audit-$(date -u +%Y%m%dT%H%M%SZ)"
mkdir -p "$OUT"

say(){ printf '\n==== %s ====\n' "$*"; }
die(){ echo "FAIL: $*" >&2; exit 1; }
command -v gcloud >/dev/null || die "gcloud is required"
command -v curl >/dev/null || die "curl is required"
command -v jq >/dev/null || die "jq is required"
command -v sha256sum >/dev/null || command -v shasum >/dev/null || die "sha256sum or shasum is required"
sha256(){ if command -v sha256sum >/dev/null; then sha256sum "$1"|awk '{print $1}'; else shasum -a 256 "$1"|awk '{print $1}'; fi; }

say "Identity / project"
ACCOUNT="$(gcloud auth list --filter=status:ACTIVE --format='value(account)' | head -1)"
[ -n "$ACCOUNT" ] || die "No active Google Cloud account"
echo "Active account: $ACCOUNT"
gcloud config set project "$PROJECT" >/dev/null
PROJECT_NUMBER="$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')"
echo "Project: $PROJECT"
echo "Project number: $PROJECT_NUMBER"
[ "$PROJECT_NUMBER" = "$PROJECT_NUMBER_EXPECTED" ] || die "Unexpected Firebase project number"

TOKEN="$(gcloud auth print-access-token)"
api_get(){
  local url="$1" dest="$2"
  local code
  code="$(curl -sS -o "$dest" -w '%{http_code}' -H "Authorization: Bearer $TOKEN" "$url")"
  if [ "$code" != "200" ]; then
    echo "HTTP $code for $url" >&2
    cat "$dest" >&2
    return 1
  fi
}

say "Firebase Rules releases"
api_get "https://firebaserules.googleapis.com/v1/projects/$PROJECT/releases?pageSize=100" "$OUT/rules-releases.json"
jq -r '.releases[]? | [.name,.rulesetName,.updateTime] | @tsv' "$OUT/rules-releases.json" | tee "$OUT/rules-releases.tsv"

compare_rule(){
  local release_suffix="$1" local_file="$2" label="$3"
  local ruleset
  ruleset="$(jq -r --arg suffix "$release_suffix" '.releases[]? | select(.name|endswith($suffix)) | .rulesetName' "$OUT/rules-releases.json" | head -1)"
  if [ -z "$ruleset" ] || [ "$ruleset" = "null" ]; then
    echo "WARN: no $label release matched suffix $release_suffix"
    return 0
  fi
  local id="${ruleset##*/}" json="$OUT/$label-ruleset.json" live="$OUT/$label-live.rules"
  api_get "https://firebaserules.googleapis.com/v1/projects/$PROJECT/rulesets/$id" "$json"
  jq -r '.source.files[0].content // empty' "$json" > "$live"
  echo "$label ruleset: $ruleset"
  echo "$label live SHA256: $(sha256 "$live")"
  if [ -f "$local_file" ]; then
    echo "$label local SHA256: $(sha256 "$local_file")"
    if cmp -s "$local_file" "$live"; then echo "PASS: $label live rules exactly match $local_file"; else echo "DIFF: $label live rules do not exactly match $local_file"; fi
  else
    echo "NOTE: local file $local_file not present in current Cloud Shell directory; live copy saved for later comparison."
  fi
}
compare_rule "/cloud.firestore" "firestore.rules" "firestore"
STORAGE_SUFFIX="/firebase.storage/fidunio-fef13.firebasestorage.app"
compare_rule "$STORAGE_SUFFIX" "storage.rules" "storage"

say "Firestore composite indexes"
gcloud firestore indexes composite list --database='(default)' --format=json > "$OUT/firestore-indexes.json"
jq -r '.[]? | [.name,.state,(.queryScope//""),(.fields|tostring)] | @tsv' "$OUT/firestore-indexes.json" | tee "$OUT/firestore-indexes.tsv"

say "Gen 2 Cloud Functions"
gcloud functions list --gen2 --regions="$REGION" --format=json > "$OUT/functions.json"
jq -r '.[]? | [.name,.state,(.serviceConfig.serviceAccountEmail//""),(.updateTime//"")] | @tsv' "$OUT/functions.json" | tee "$OUT/functions.tsv"
EXPECTED_FUNCTIONS=(
 enrollRecoveryV1 startE2EERecoveryV1 completeE2EERecoveryV1
 createAdminRecoveryAuthorizationV1 listAdminRecoveryAuthorizationsV1 revokeAdminRecoveryAuthorizationV1
 startAdminAuthorizedRecoveryV1 completeAdminAuthorizedRecoveryV1
 getLegalAcceptanceV1 acceptLegalPolicyV1 completeMyAccountDeletionV1 transferSystemOwnershipV1
 deleteDirectMessageForEveryoneV1 deleteMyMessagesForEveryoneV1 deleteConversationForEveryoneV1
 notifyDirectMessageCreatedV1 notifyGroupMessageCreatedV1 purgeDisappearingMessagesV1
)
for fn in "${EXPECTED_FUNCTIONS[@]}"; do
  if jq -e --arg n "$fn" '.[]? | select((.name|split("/")|last)==$n)' "$OUT/functions.json" >/dev/null; then
    echo "PASS function present: $fn"
  else
    echo "MISSING function: $fn"
  fi
done

say "Function service-account authority"
jq -r '.[]? | [(.name|split("/")|last),(.serviceConfig.serviceAccountEmail//"MISSING")] | @tsv' "$OUT/functions.json" | tee "$OUT/function-service-accounts.tsv"

say "Firebase App Check iOS App Attest registration"
if api_get "https://firebaseappcheck.googleapis.com/v1/projects/$PROJECT_NUMBER/apps/$IOS_APP_ID/appAttestConfig" "$OUT/appcheck-ios-appattest.json"; then
  jq '{name,tokenTtl}' "$OUT/appcheck-ios-appattest.json"
  echo "PASS: iOS App Attest configuration exists"
else
  echo "MISSING/UNREADABLE: iOS App Attest configuration"
fi

say "Firebase App Check web reCAPTCHA Enterprise registration"
if api_get "https://firebaseappcheck.googleapis.com/v1/projects/$PROJECT_NUMBER/apps/$WEB_APP_ID/recaptchaEnterpriseConfig" "$OUT/appcheck-web-enterprise.json"; then
  jq '{name,tokenTtl,siteKey,riskAnalysis}' "$OUT/appcheck-web-enterprise.json"
  echo "PASS: web reCAPTCHA Enterprise configuration exists"
else
  echo "MISSING/UNREADABLE: web reCAPTCHA Enterprise configuration"
fi

say "Firebase App Check enforcement state"
api_get "https://firebaseappcheck.googleapis.com/v1/projects/$PROJECT_NUMBER/services?pageSize=100" "$OUT/appcheck-services.json"
jq -r '.services[]? | [.name,.enforcementMode,(.replayProtection//"OFF"),(.updateTime//"")] | @tsv' "$OUT/appcheck-services.json" | tee "$OUT/appcheck-services.tsv"
if jq -e '.services[]? | select(.enforcementMode=="ENFORCED")' "$OUT/appcheck-services.json" >/dev/null; then
  echo "ATTENTION: at least one App Check service is ENFORCED. Approved FIDUNIO standby policy expects enforcement OFF unless deliberately activated."
else
  echo "PASS: no returned App Check service is in ENFORCED mode"
fi

say "Firebase app registrations"
if api_get "https://firebase.googleapis.com/v1beta1/projects/$PROJECT/iosApps?pageSize=100" "$OUT/firebase-ios-apps.json"; then
  jq -r '.apps[]? | [.appId,.bundleId,.displayName] | @tsv' "$OUT/firebase-ios-apps.json" | tee "$OUT/firebase-ios-apps.tsv"
fi
if api_get "https://firebase.googleapis.com/v1beta1/projects/$PROJECT/webApps?pageSize=100" "$OUT/firebase-web-apps.json"; then
  jq -r '.apps[]? | [.appId,.displayName] | @tsv' "$OUT/firebase-web-apps.json" | tee "$OUT/firebase-web-apps.tsv"
fi

say "Audit complete"
echo "READ-ONLY audit output: $OUT"
echo "No Firebase/Google Cloud resource was changed by this script."
