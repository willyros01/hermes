#!/usr/bin/env bash
# FIDUNIO Amazon ADM — one-time secure GitHub-to-Firebase connection.
# Upload this TXT into Google Cloud Shell, then run: bash fsec.txt
# NEVER type or paste your Amazon credentials here. They're in GitHub Actions.
# Fixed project, branch, service account, secret names. No Firebase deployment.
set -Eeuo pipefail
umask 077
P='fidunio-fef13'
NUM='130339622893'
POOL='fidunio-adm-sync'
PROV='hermes-android'
SA="fidunio-adm-sync@$P.iam.gserviceaccount.com"
ONE='FIDUNIO_ADM_CLIENT_ID'
TWO='FIDUNIO_ADM_CLIENT_SECRET'
BACKUP="$HOME/fsbk.txt"
if [ -e "$BACKUP" ]; then
  BACKUP="$HOME/fsbk-$(date -u +%Y%m%d%H%M%S).txt"
fi
echo "FIDUNIO private ADM sync bootstrap — $P"
echo 'This creates a narrowly authorized connection. No passwords are entered.'
command -v gcloud >/dev/null || { echo 'ERROR: Google Cloud Shell required.'; exit 1; }
test "$(gcloud projects describe "$P" --format='value(projectNumber)')" = "$NUM" || { echo 'ERROR: Wrong project number.'; exit 1; }
test -n "$(gcloud auth list --filter='status:ACTIVE' --format='value(account)' | head -n1)" || { echo 'ERROR: No active Google Cloud account.'; exit 1; }

# Check existing resources without reading any secret payloads.
for NAME in "$ONE" "$TWO"; do
  if gcloud secrets describe "$NAME" --project="$P" >/dev/null 2>&1; then
    echo "STOP: $NAME already exists; will not edit an existing secret."
    exit 2
  fi
done
if gcloud iam workload-identity-pools describe "$POOL" --location=global --project="$P" >/dev/null 2>&1; then
  echo 'STOP: dedicated identity pool already exists. Inspect before retrying.'
  exit 2
fi
if gcloud iam service-accounts describe "$SA" --project="$P" >/dev/null 2>&1; then
  echo 'STOP: dedicated service account already exists. Inspect before retrying.'
  exit 2
fi
{
  date -u '+Firebase PRE-CHANGE snapshot UTC %Y-%m-%dT%H:%M:%SZ'
  echo "project=$P number=$NUM"
  echo 'GitHub repository=willyros01/hermes; allowed ref=refs/heads/Android'
  echo "$ONE: absent"
  echo "$TWO: absent"
  echo "WIF pool $POOL: absent"
  echo "Service account $SA: absent"
  echo
  echo 'Rollback: only after confirming no ADM sender functions have been deployed.'
  echo "  gcloud iam service-accounts delete $SA --project=$P"
  echo "  gcloud iam workload-identity-pools delete $POOL --location=global --project=$P"
  echo "  gcloud secrets delete $ONE --project=$P"
  echo "  gcloud secrets delete $TWO --project=$P"
  echo 'Existing Firestore data, web/iOS functions, IAM principals and storage are not modified.'
} > "$BACKUP"
chmod 600 "$BACKUP"
echo "Pre-change snapshot: $BACKUP"

# Enable API only if currently disabled (report without hiding errors).
gcloud services enable iam.googleapis.com iamcredentials.googleapis.com sts.googleapis.com secretmanager.googleapis.com --project="$P" --quiet

# Separate pool prevents Android writers from inheriting iOS read-only audit grants.
gcloud iam workload-identity-pools create "$POOL" --project="$P" --location=global --display-name='FIDUNIO ADM secrets sync'
gcloud iam workload-identity-pools providers create-oidc "$PROV" \
  --project="$P" --location=global --workload-identity-pool="$POOL" \
  --display-name='Hermes Android OIDC only' \
  --issuer-uri='https://token.actions.githubusercontent.com' \
  --attribute-mapping='google.subject=assertion.sub,attribute.repository=assertion.repository,attribute.ref=assertion.ref' \
  --attribute-condition="assertion.repository=='willyros01/hermes' && assertion.ref=='refs/heads/Android'"
gcloud iam service-accounts create 'fidunio-adm-sync' --project="$P" --display-name='FIDUNIO ADM secret version writer'
gcloud iam service-accounts add-iam-policy-binding "$SA" --project="$P" \
  --role='roles/iam.workloadIdentityUser' \
  --member="principalSet://iam.googleapis.com/projects/$NUM/locations/global/workloadIdentityPools/$POOL/attribute.repository/willyros01/hermes" >/dev/null

# Create blank secrets only. Actual values arrive later directly from GitHub
# Actions via OIDC; they are never present in this Cloud Shell or in source.
for NAME in "$ONE" "$TWO"; do
  gcloud secrets create "$NAME" --replication-policy=automatic --project="$P"
  gcloud secrets add-iam-policy-binding "$NAME" --project="$P" \
    --member="serviceAccount:$SA" \
    --role='roles/secretmanager.secretVersionAdder' >/dev/null
done
echo 'PASS: Scoped GitHub-to-Firebase identity and empty secret containers configured.'
echo "Backup: $BACKUP"
echo 'No secrets were displayed, copied, or deployed.'
echo 'Next: authorized GitHub Actions transfers the two values into Secret Manager.'
