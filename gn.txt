#!/usr/bin/env bash
set -euo pipefail

PROJECT="fidunio-fef13"
REGION="us-central1"
FIRESTORE_REGION="nam5"
FUNCTION="notifyGroupMessageCreatedV1"
CODEBASE="recovery"
SERVICE_ACCOUNT="fidunio-notification@fidunio-fef13.iam.gserviceaccount.com"
COMMIT="c273f0bfc0275c612fb91c916da0f4d08109dcaa"
SCRIPT_VERSION="1.1.28-a"

ACTIVE_PROJECT="$(gcloud config get-value project 2>/dev/null)"
if [[ "$ACTIVE_PROJECT" != "$PROJECT" ]]; then
  echo "STOP: Cloud Shell project is '$ACTIVE_PROJECT', expected '$PROJECT'."
  exit 1
fi

if ! command -v firebase >/dev/null 2>&1; then
  echo "Firebase CLI not found; installing firebase-tools..."
  npm install -g firebase-tools
fi

if ! gcloud iam service-accounts describe "$SERVICE_ACCOUNT" --project="$PROJECT" >/dev/null 2>&1; then
  echo "STOP: dedicated notification service account is missing."
  exit 1
fi

WORK="$(mktemp -d)"
trap 'rm -rf "$WORK"' EXIT
ROOT="$WORK/fidunio"
SOURCE="$ROOT/functions"
mkdir -p "$SOURCE/recovery" "$SOURCE/message-delete" "$SOURCE/disappearing" "$SOURCE/notification"
RAW="https://raw.githubusercontent.com/willyros01/hermes/$COMMIT"

echo "FIDUNIO 1.1.28 group-notification backend deployment"
echo "Project: $PROJECT"
echo "Function: $FUNCTION"
echo "Pinned source: $COMMIT"
echo "Script: $SCRIPT_VERSION"

download_root(){
  local file="$1"
  curl --retry 3 --retry-delay 2 -fsSL "$RAW/$file" -o "$ROOT/$file"
}
download_function(){
  local file="$1"
  curl --retry 3 --retry-delay 2 -fsSL "$RAW/functions/$file" -o "$SOURCE/$file"
}

download_root "firebase.json"
download_function "package.json"
download_function "package-lock.json"
download_function "index.mjs"
download_function "recovery/e2ee-recovery-callable-core.mjs"
download_function "recovery/e2ee-recovery-firestore-admin-adapter.mjs"
download_function "recovery/e2ee-recovery-server-crypto.mjs"
download_function "recovery/e2ee-recovery-session-policy.mjs"
download_function "message-delete/message-delete-callable-core.mjs"
download_function "message-delete/message-delete-firestore-admin-adapter.mjs"
download_function "disappearing/disappearing-content-policy.js"
download_function "disappearing/disappearing-group-grant-trace-plan.js"
download_function "disappearing/disappearing-purge-policy.js"
download_function "disappearing/disappearing-purge-executor.js"
download_function "disappearing/disappearing-purge-firestore-admin-adapter.mjs"
download_function "disappearing/disappearing-scheduler-core.mjs"
download_function "notification/direct-message-notification-core.mjs"
download_function "notification/direct-message-notification-firestore-admin-adapter.mjs"
download_function "notification/group-message-notification-core.mjs"

if ! grep -q 'groups/{groupId}/messages/{messageId}' "$SOURCE/index.mjs"; then
  echo "STOP: pinned source does not contain the reviewed group-message trigger."
  exit 1
fi
if ! grep -q 'type:"group-message"' "$SOURCE/notification/group-message-notification-core.mjs"; then
  echo "STOP: pinned source does not contain the reviewed data-only group payload."
  exit 1
fi

echo "Preflighting the complete pinned Functions module graph..."
npm --prefix "$SOURCE" install --ignore-scripts --silent
(
  cd "$ROOT"
  node --input-type=module -e "const m=await import('./functions/index.mjs');if(typeof m.notifyGroupMessageCreatedV1!=='function')throw new Error('group notification export missing');console.log('Pinned Functions graph imports successfully')"
)

cd "$ROOT"
firebase deploy \
  --project "$PROJECT" \
  --only "functions:$CODEBASE:$FUNCTION" \
  --non-interactive

STATE="$(gcloud functions describe "$FUNCTION" --gen2 --project="$PROJECT" --region="$REGION" --format='value(state)')"
RUNTIME_SA="$(gcloud functions describe "$FUNCTION" --gen2 --project="$PROJECT" --region="$REGION" --format='value(serviceConfig.serviceAccountEmail)')"
RUN_RESOURCE="$(gcloud functions describe "$FUNCTION" --gen2 --project="$PROJECT" --region="$REGION" --format='value(serviceConfig.service)')"
RUN_SERVICE="${RUN_RESOURCE##*/}"
if [[ "$STATE" != "ACTIVE" ]]; then
  echo "STOP: Function deployment state is '$STATE'."
  exit 1
fi
if [[ "$RUNTIME_SA" != "$SERVICE_ACCOUNT" ]]; then
  echo "STOP: Function runtime service account is '$RUNTIME_SA', expected '$SERVICE_ACCOUNT'."
  exit 1
fi
if [[ -z "$RUN_SERVICE" ]]; then
  echo "STOP: Cloud Run service could not be resolved."
  exit 1
fi

TRIGGER_ROW="$(gcloud eventarc triggers list --project="$PROJECT" --location="$FIRESTORE_REGION" --format='csv[no-heading](name,serviceAccount,destination.cloudRun.service)' | grep -i "${FUNCTION,,}" | head -n 1 || true)"
if [[ -z "$TRIGGER_ROW" ]]; then
  echo "STOP: Eventarc trigger for $FUNCTION was not found in $FIRESTORE_REGION."
  exit 1
fi
IFS=',' read -r TRIGGER TRIGGER_SA TRIGGER_SERVICE <<< "$TRIGGER_ROW"
if [[ -z "$TRIGGER" || -z "$TRIGGER_SA" ]]; then
  echo "STOP: Eventarc trigger identity could not be resolved."
  exit 1
fi

gcloud run services add-iam-policy-binding "$RUN_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --member="serviceAccount:$TRIGGER_SA" \
  --role="roles/run.invoker" \
  --quiet >/dev/null

if ! gcloud run services get-iam-policy "$RUN_SERVICE" --project="$PROJECT" --region="$REGION" --format=json | jq -e --arg member "serviceAccount:$TRIGGER_SA" '.bindings[] | select(.role=="roles/run.invoker") | .members[] | select(.==$member)' >/dev/null; then
  echo "STOP: exact Eventarc-to-Cloud-Run invocation binding was not verified."
  exit 1
fi

echo "SUCCESS"
echo "Function: $FUNCTION ($STATE)"
echo "Runtime SA: $RUNTIME_SA"
echo "Eventarc trigger: $TRIGGER ($FIRESTORE_REGION)"
echo "Trigger SA: $TRIGGER_SA"
echo "Cloud Run service: $RUN_SERVICE"
echo "Pinned source: $COMMIT"
echo "No Firestore rules or other Functions were deployed."
