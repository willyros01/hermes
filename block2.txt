#!/usr/bin/env bash
# FIDUNIO Block receipt barrier repair
# ONE COMMAND, NO PARAMETERS:
#   bash block2.txt
#
# Deploys ONLY the minimal Firestore rules change that prevents Delivered/Read
# receipt advancement after either participant blocks the other.
set -euo pipefail

PROJECT="fidunio-fef13"
PROJECT_NUMBER_EXPECTED="130339622893"
REPO="willyros01/hermes"
PINNED_COMMIT="035aeb5237377a643e5d333fef4e755b5d32523e"
LIVE_BASE_PATH="build/firestore-safety-rules-candidate.rules"
CANDIDATE_PATH="build/firestore-block-receipt-rules-candidate.rules"
RULES_API="https://firebaserules.googleapis.com/v1"

say(){ printf '\n==== %s ====\n' "$*"; }
stop(){ echo "STOP: $*" >&2; exit 1; }

for cmd in gcloud curl jq python3 cmp tar npm; do command -v "$cmd" >/dev/null 2>&1 || stop "$cmd is required"; done

say "FIDUNIO project safety switch"
gcloud config set project "$PROJECT" >/dev/null
ACTIVE="$(gcloud config get-value project 2>/dev/null)"
NUMBER="$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')"
[[ "$ACTIVE" == "$PROJECT" ]] || stop "Wrong active project '$ACTIVE'."
[[ "$NUMBER" == "$PROJECT_NUMBER_EXPECTED" ]] || stop "Wrong project number '$NUMBER'."
echo "PASS: target project is $PROJECT ($NUMBER)."

STAMP="$(date -u +%Y%m%dT%H%M%SZ)"
BACKUP_DIR="$HOME/fidunio-block-receipt-rules-$STAMP"
WORK="$(mktemp -d)"
mkdir -p "$BACKUP_DIR"
trap 'rm -rf "$WORK"' EXIT

say "Capture exact current live Firestore rules"
TOKEN="$(gcloud auth print-access-token)"
curl -fsS -H "Authorization: Bearer $TOKEN" "$RULES_API/projects/$PROJECT/releases/cloud.firestore" -o "$BACKUP_DIR/release-before.json"
RULESET_NAME="$(jq -r '.rulesetName // empty' "$BACKUP_DIR/release-before.json")"
[[ "$RULESET_NAME" == projects/*/rulesets/* ]] || stop "Could not determine current Firestore ruleset."
curl -fsS -H "Authorization: Bearer $TOKEN" "$RULES_API/$RULESET_NAME" -o "$BACKUP_DIR/ruleset-before.json"
python3 - "$BACKUP_DIR/ruleset-before.json" "$BACKUP_DIR/firestore-before.rules" <<'PY'
import json,sys
src=json.load(open(sys.argv[1],encoding="utf-8"))
files=src.get("source",{}).get("files",[])
if len(files)!=1: raise SystemExit(f"STOP: expected one live Firestore rules file, found {len(files)}")
content=files[0].get("content")
if not isinstance(content,str) or not content.strip(): raise SystemExit("STOP: live Firestore rules are empty")
open(sys.argv[2],"w",encoding="utf-8",newline="").write(content)
PY

BASE_URL="https://raw.githubusercontent.com/$REPO/$PINNED_COMMIT/$LIVE_BASE_PATH"
CANDIDATE_URL="https://raw.githubusercontent.com/$REPO/$PINNED_COMMIT/$CANDIDATE_PATH"
curl --retry 3 --retry-delay 2 -fsSL "$BASE_URL" -o "$WORK/base.rules"
curl --retry 3 --retry-delay 2 -fsSL "$CANDIDATE_URL" -o "$WORK/candidate.rules"

say "Safety check: live rules must still equal verified Block + Abuse base"
if ! cmp -s "$BACKUP_DIR/firestore-before.rules" "$WORK/base.rules"; then
  echo "NO DEPLOYMENT WAS PERFORMED."
  stop "Live rules changed unexpectedly. Refusing to overwrite them."
fi
echo "PASS: live rules exactly match the previously verified safety rules."

say "Prove candidate changes only the direct receipt block barrier"
python3 - "$WORK/base.rules" "$WORK/candidate.rules" <<'PY'
import sys
base=open(sys.argv[1],encoding="utf-8").read()
candidate=open(sys.argv[2],encoding="utf-8").read()
old='function validDirectReceiptUpdate(id,d,old){return request.auth.uid!=old.senderUid&&request.auth.uid in conversationDoc(id).data.members&&d.senderUid==old.senderUid'
new='function validDirectReceiptUpdate(id,d,old){return request.auth.uid!=old.senderUid&&request.auth.uid in conversationDoc(id).data.members&&directPairUnblocked(id)&&d.senderUid==old.senderUid'
if old not in base: raise SystemExit("STOP: expected receipt authority is missing from verified base")
expected=base.replace(old,new,1)
if candidate != expected: raise SystemExit("STOP: candidate contains changes beyond the receipt block barrier")
print("PASS: candidate differs only by directPairUnblocked receipt enforcement.")
PY

say "Prepare exact candidate test workspace"
ARCHIVE="$WORK/repo.tar.gz"
curl --retry 3 --retry-delay 2 -fsSL "https://github.com/$REPO/archive/$PINNED_COMMIT.tar.gz" -o "$ARCHIVE"
tar -xzf "$ARCHIVE" -C "$WORK"
ROOT="$WORK/hermes-$PINNED_COMMIT"
[[ -d "$ROOT" ]] || stop "Pinned workspace did not unpack."
cp "$WORK/candidate.rules" "$ROOT/firestore.rules"
if ! command -v firebase >/dev/null 2>&1; then npm install -g firebase-tools; fi
(cd "$ROOT" && npm ci --ignore-scripts --no-audit --no-fund >/dev/null)

say "Run Block/Unblock + blocked receipt/reaction regression"
(cd "$ROOT" && firebase emulators:exec --only firestore --project demo-fidunio-block-rules "node firestore-user-block.rules.test.mjs")

say "Re-run Report Abuse regression to prove preservation"
(cd "$ROOT" && firebase emulators:exec --only firestore --project demo-fidunio-abuse-report-rules "node firestore-abuse-report.rules.test.mjs")

say "Re-confirm project immediately before deployment"
gcloud config set project "$PROJECT" >/dev/null
[[ "$(gcloud config get-value project 2>/dev/null)" == "$PROJECT" ]] || stop "Project changed."
[[ "$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')" == "$PROJECT_NUMBER_EXPECTED" ]] || stop "Project number changed."

say "Prepare exact rollback"
cp "$BACKUP_DIR/firestore-before.rules" "$BACKUP_DIR/firestore.rules"
cat > "$BACKUP_DIR/firebase.json" <<'JSON'
{"firestore":{"rules":"firestore.rules"}}
JSON
cat > "$BACKUP_DIR/rollback.sh" <<ROLLBACK
#!/usr/bin/env bash
set -euo pipefail
gcloud config set project "$PROJECT" >/dev/null
[[ "\$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')" == "$PROJECT_NUMBER_EXPECTED" ]] || exit 1
cd "$BACKUP_DIR"
firebase deploy --only firestore:rules --project "$PROJECT" --non-interactive
echo "ROLLBACK COMPLETE: exact pre-change Firestore rules restored."
ROLLBACK
chmod +x "$BACKUP_DIR/rollback.sh"
echo "Rollback prepared: $BACKUP_DIR/rollback.sh"

say "Deploy ONLY tested Firestore rules"
DEPLOY="$WORK/deploy"; mkdir -p "$DEPLOY"
cp "$WORK/candidate.rules" "$DEPLOY/firestore.rules"
printf '%s\n' '{"firestore":{"rules":"firestore.rules"}}' > "$DEPLOY/firebase.json"
(cd "$DEPLOY" && firebase deploy --only firestore:rules --project "$PROJECT" --non-interactive)

say "Verify exact tested candidate is live"
sleep 3
TOKEN="$(gcloud auth print-access-token)"
curl -fsS -H "Authorization: Bearer $TOKEN" "$RULES_API/projects/$PROJECT/releases/cloud.firestore" -o "$BACKUP_DIR/release-after.json"
NEW_RULESET="$(jq -r '.rulesetName // empty' "$BACKUP_DIR/release-after.json")"
[[ "$NEW_RULESET" == projects/*/rulesets/* ]] || stop "Could not determine post-deploy ruleset."
curl -fsS -H "Authorization: Bearer $TOKEN" "$RULES_API/$NEW_RULESET" -o "$BACKUP_DIR/ruleset-after.json"
python3 - "$BACKUP_DIR/ruleset-after.json" "$BACKUP_DIR/firestore-after.rules" <<'PY'
import json,sys
src=json.load(open(sys.argv[1],encoding="utf-8"));files=src.get("source",{}).get("files",[])
if len(files)!=1: raise SystemExit("STOP: deployed rules source shape changed")
open(sys.argv[2],"w",encoding="utf-8",newline="").write(files[0].get("content",""))
PY
cmp -s "$BACKUP_DIR/firestore-after.rules" "$WORK/candidate.rules" || stop "Live rules differ from tested candidate. Use rollback: $BACKUP_DIR/rollback.sh"

echo
echo "==== SUCCESS ===="
echo "FIDUNIO blocked direct receipt barrier is live."
echo "Project: $PROJECT ($PROJECT_NUMBER_EXPECTED)"
echo "Previous ruleset: $RULESET_NAME"
echo "New ruleset: $NEW_RULESET"
echo "Rollback directory: $BACKUP_DIR"
echo "No Functions, IAM, App Check, Storage rules, application data, or TestFlight build were changed."
