#!/usr/bin/env bash
# FIDUNIO Block / Unblock Firestore rules repair
#
# ONE COMMAND, NO PARAMETERS:
#   bash blk.txt
#
# This script:
# - explicitly switches to fidunio-fef13
# - verifies project number 130339622893
# - downloads and saves the exact currently-live Firestore rules
# - refuses to continue unless live rules exactly match protected production main
# - tests the production-based Block/Unblock candidate in the Firestore emulator
# - deploys ONLY Firestore rules
# - verifies the exact candidate became live
# - writes a rollback script that restores the exact pre-change live rules
#
# It does NOT build TestFlight, deploy Functions, change IAM, change App Check,
# change Storage rules, or modify Firestore application data.
set -euo pipefail

PROJECT="fidunio-fef13"
PROJECT_NUMBER_EXPECTED="130339622893"
PRODUCTION_MAIN_SHA="8ce96f85f132da078bbfc237825e78f139deea96"
CANDIDATE_COMMIT="bb5f76f3ec6ec66f88aab748cd8bf681f7888121"
REPO="willyros01/hermes"
RULES_API="https://firebaserules.googleapis.com/v1"
SCRIPT_VERSION="block-rules-repair-1"

say(){ printf '\n==== %s ====\n' "$*"; }
stop(){ echo "STOP: $*" >&2; exit 1; }

for cmd in gcloud curl jq python3 diff tar npm; do
  command -v "$cmd" >/dev/null 2>&1 || stop "$cmd is required"
done

say "FIDUNIO project safety switch"
BEFORE="$(gcloud config get-value project 2>/dev/null || true)"
echo "Project before switch: ${BEFORE:-<none>}"
echo "Switching explicitly to: $PROJECT"
gcloud config set project "$PROJECT" >/dev/null
ACTIVE="$(gcloud config get-value project 2>/dev/null)"
[[ "$ACTIVE" == "$PROJECT" ]] || stop "Project switch failed; active project is '$ACTIVE'."
NUMBER="$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')"
[[ "$NUMBER" == "$PROJECT_NUMBER_EXPECTED" ]] || stop "Wrong project number '$NUMBER'."
echo "PASS: target project is $PROJECT ($NUMBER)."

STAMP="$(date -u +%Y%m%dT%H%M%SZ)"
BACKUP_DIR="$HOME/fidunio-block-rules-$STAMP"
WORK="$(mktemp -d)"
mkdir -p "$BACKUP_DIR"
trap 'rm -rf "$WORK"' EXIT

say "Capture exact current live Firestore rules"
TOKEN="$(gcloud auth print-access-token)"
curl -fsS \
  -H "Authorization: Bearer $TOKEN" \
  "$RULES_API/projects/$PROJECT/releases/cloud.firestore" \
  -o "$BACKUP_DIR/release-before.json"

RULESET_NAME="$(jq -r '.rulesetName // empty' "$BACKUP_DIR/release-before.json")"
[[ "$RULESET_NAME" == projects/*/rulesets/* ]] || stop "Could not determine current Firestore ruleset."
echo "$RULESET_NAME" > "$BACKUP_DIR/ruleset-before.txt"
echo "Current ruleset: $RULESET_NAME"

curl -fsS \
  -H "Authorization: Bearer $TOKEN" \
  "$RULES_API/$RULESET_NAME" \
  -o "$BACKUP_DIR/ruleset-before.json"

python3 - "$BACKUP_DIR/ruleset-before.json" "$BACKUP_DIR/firestore-before.rules" <<'PY'
import json,sys
src=json.load(open(sys.argv[1],encoding="utf-8"))
files=src.get("source",{}).get("files",[])
if len(files)!=1:
    raise SystemExit(f"STOP: expected one live Firestore rules source file, found {len(files)}")
content=files[0].get("content")
if not isinstance(content,str) or not content.strip():
    raise SystemExit("STOP: live Firestore rules source is empty")
open(sys.argv[2],"w",encoding="utf-8",newline="").write(content)
print("Saved live rules:",sys.argv[2])
PY

say "Download protected production baseline and isolated candidate"
MAIN_URL="https://raw.githubusercontent.com/$REPO/$PRODUCTION_MAIN_SHA/firestore.rules"
CANDIDATE_URL="https://raw.githubusercontent.com/$REPO/$CANDIDATE_COMMIT/build/firestore-block-rules-candidate.rules"
curl --retry 3 --retry-delay 2 -fsSL "$MAIN_URL" -o "$WORK/main.rules"
curl --retry 3 --retry-delay 2 -fsSL "$CANDIDATE_URL" -o "$WORK/candidate.rules"

say "Safety check: live rules must equal protected production main"
if ! cmp -s "$BACKUP_DIR/firestore-before.rules" "$WORK/main.rules"; then
  echo "The current live Firestore rules do NOT exactly match protected production main."
  echo "NO DEPLOYMENT WAS PERFORMED."
  echo "Difference saved at: $BACKUP_DIR/live-vs-main.diff"
  diff -u "$WORK/main.rules" "$BACKUP_DIR/firestore-before.rules" > "$BACKUP_DIR/live-vs-main.diff" || true
  stop "Refusing to overwrite an unexpected live rules state."
fi
echo "PASS: live Firestore rules exactly match production main $PRODUCTION_MAIN_SHA."

say "Verify candidate contains only the intended Block/Unblock authority"
grep -Fq 'function blockPath(ownerUid,targetUid)' "$WORK/candidate.rules" || stop "Candidate missing blockPath."
grep -Fq 'function directPairUnblocked(id)' "$WORK/candidate.rules" || stop "Candidate missing directPairUnblocked."
grep -Fq 'function directCreatePairUnblocked(d)' "$WORK/candidate.rules" || stop "Candidate missing directCreatePairUnblocked."
grep -Fq 'match /blocks/{blockedUid}' "$WORK/candidate.rules" || stop "Candidate missing Block/Unblock rules."
grep -Fq 'allow delete: if registered()&&request.auth.uid==uid;' "$WORK/candidate.rules" || stop "Candidate missing Unblock delete authority."
grep -Fq 'directCreatePairUnblocked(request.resource.data)' "$WORK/candidate.rules" || stop "Candidate does not block new direct conversations."
grep -Fq 'directPairUnblocked(conversationId)' "$WORK/candidate.rules" || stop "Candidate does not block direct sends."
if grep -Fq 'match /abuseReports/{reportId}' "$WORK/candidate.rules"; then
  stop "Candidate unexpectedly contains unrelated Abuse Reports rules."
fi
echo "PASS: candidate is limited to the Block/Unblock security boundary."

say "Prepare exact candidate test workspace"
ARCHIVE="$WORK/repo.tar.gz"
curl --retry 3 --retry-delay 2 -fsSL \
  "https://github.com/$REPO/archive/$CANDIDATE_COMMIT.tar.gz" \
  -o "$ARCHIVE"
tar -xzf "$ARCHIVE" -C "$WORK"
ROOT="$WORK/hermes-$CANDIDATE_COMMIT"
[[ -d "$ROOT" ]] || stop "Pinned test workspace did not unpack."
cp "$WORK/candidate.rules" "$ROOT/firestore.rules"

if ! command -v firebase >/dev/null 2>&1; then
  echo "Firebase CLI not found; installing firebase-tools..."
  npm install -g firebase-tools
fi

say "Run Block/Unblock Firestore emulator acceptance"
(
  cd "$ROOT"
  npm ci --ignore-scripts --no-audit --no-fund >/dev/null
  firebase emulators:exec \
    --only firestore \
    --project demo-fidunio-block-rules \
    "node firestore-user-block.rules.test.mjs"
)

say "Re-confirm project immediately before live rules deployment"
gcloud config set project "$PROJECT" >/dev/null
ACTIVE="$(gcloud config get-value project 2>/dev/null)"
NUMBER="$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')"
[[ "$ACTIVE" == "$PROJECT" ]] || stop "Active project changed unexpectedly."
[[ "$NUMBER" == "$PROJECT_NUMBER_EXPECTED" ]] || stop "Project number changed unexpectedly."
echo "PASS: live deployment target is $PROJECT ($NUMBER)."

say "Create rollback script before deployment"
cp "$BACKUP_DIR/firestore-before.rules" "$BACKUP_DIR/firestore.rules"
cat > "$BACKUP_DIR/firebase.json" <<'JSON'
{
  "firestore": {
    "rules": "firestore.rules"
  }
}
JSON
cat > "$BACKUP_DIR/rollback.sh" <<ROLLBACK
#!/usr/bin/env bash
set -euo pipefail
PROJECT="$PROJECT"
EXPECTED="$PROJECT_NUMBER_EXPECTED"
DIR="$BACKUP_DIR"

gcloud config set project "\$PROJECT" >/dev/null
ACTIVE="\$(gcloud config get-value project 2>/dev/null)"
NUMBER="\$(gcloud projects describe "\$PROJECT" --format='value(projectNumber)')"
[[ "\$ACTIVE" == "\$PROJECT" ]] || { echo "STOP: wrong project"; exit 1; }
[[ "\$NUMBER" == "\$EXPECTED" ]] || { echo "STOP: wrong project number"; exit 1; }

cd "\$DIR"
firebase deploy --only firestore:rules --project "\$PROJECT" --non-interactive
echo "ROLLBACK COMPLETE: restored exact pre-change Firestore rules."
ROLLBACK
chmod +x "$BACKUP_DIR/rollback.sh"
echo "Rollback prepared: $BACKUP_DIR/rollback.sh"

say "Deploy ONLY the tested Firestore rules candidate"
DEPLOY="$WORK/deploy"
mkdir -p "$DEPLOY"
cp "$WORK/candidate.rules" "$DEPLOY/firestore.rules"
cat > "$DEPLOY/firebase.json" <<'JSON'
{
  "firestore": {
    "rules": "firestore.rules"
  }
}
JSON
(
  cd "$DEPLOY"
  firebase deploy --only firestore:rules --project "$PROJECT" --non-interactive
)

say "Verify exact candidate is now live"
sleep 3
TOKEN="$(gcloud auth print-access-token)"
curl -fsS \
  -H "Authorization: Bearer $TOKEN" \
  "$RULES_API/projects/$PROJECT/releases/cloud.firestore" \
  -o "$BACKUP_DIR/release-after.json"
NEW_RULESET="$(jq -r '.rulesetName // empty' "$BACKUP_DIR/release-after.json")"
[[ "$NEW_RULESET" == projects/*/rulesets/* ]] || stop "Could not determine post-deploy Firestore ruleset."
curl -fsS \
  -H "Authorization: Bearer $TOKEN" \
  "$RULES_API/$NEW_RULESET" \
  -o "$BACKUP_DIR/ruleset-after.json"

python3 - "$BACKUP_DIR/ruleset-after.json" "$BACKUP_DIR/firestore-after.rules" <<'PY'
import json,sys
src=json.load(open(sys.argv[1],encoding="utf-8"))
files=src.get("source",{}).get("files",[])
if len(files)!=1:
    raise SystemExit(f"STOP: expected one deployed rules source file, found {len(files)}")
open(sys.argv[2],"w",encoding="utf-8",newline="").write(files[0].get("content",""))
PY

cmp -s "$BACKUP_DIR/firestore-after.rules" "$WORK/candidate.rules" ||
  stop "Deployment returned success, but live rules do not exactly match the tested candidate. Rollback is available at $BACKUP_DIR/rollback.sh"

echo "PASS: exact tested Block/Unblock candidate is live."
echo "New ruleset: $NEW_RULESET"

say "SUCCESS"
echo "FIDUNIO Block/Unblock Firestore repair completed."
echo "Project: $PROJECT ($PROJECT_NUMBER_EXPECTED)"
echo "Production base: $PRODUCTION_MAIN_SHA"
echo "Candidate commit: $CANDIDATE_COMMIT"
echo "Pre-change ruleset: $RULESET_NAME"
echo "Post-change ruleset: $NEW_RULESET"
echo "Rollback directory: $BACKUP_DIR"
echo
echo "No Functions, IAM, App Check, Storage rules, application data, or TestFlight build were changed."
echo
echo "NEXT DEVICE TEST:"
echo "1. FIDUNIO 1.1.72 -> Settings -> Safety -> Blocked Users."
echo "2. Choose a disposable/test FIDUNIO user and tap Block User."
echo "3. Confirm the blocked user appears in the list with an Unblock button."
echo "4. Tap Unblock and confirm the user disappears from the blocked list."
