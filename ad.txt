#!/usr/bin/env bash
# FIDUNIO self-service account deletion live preflight / controlled deploy handoff.
#
# DEFAULT BEHAVIOR IS READ-ONLY.
# This script never changes IAM, Firestore rules, App Check, Storage, data, or Auth users.
#
# Audit:
#   bash ad.txt
#
# Deploy ONLY after explicit approval in chat:
#   APPROVE_DEPLOY=YES bash ad.txt
#
# Safety:
# - Deploy occurs only when ALL FIVE deletion callables are absent.
# - If any subset exists, or any existing callable has unexpected state/runtime identity,
#   the script stops without changing anything.
# - No IAM role is added automatically.
set -euo pipefail

PROJECT="fidunio-fef13"
PROJECT_NUMBER_EXPECTED="130339622893"
REGION="us-central1"
CODEBASE="recovery"
SERVICE_ACCOUNT="fidunio-message-delete@fidunio-fef13.iam.gserviceaccount.com"
COMMIT="71104cc7499dc51143a0a3dda00cd6b1fc680825"
SCRIPT_VERSION="account-delete-preflight-2"
APPROVE_DEPLOY="${APPROVE_DEPLOY:-NO}"

FUNCTIONS=(
  getMyAccountDeletionRequestV1
  requestMyAccountDeletionV1
  markMyAccountDeletionCleanupCompleteV1
  cancelMyAccountDeletionRequestV1
  completeMyAccountDeletionV1
)

say(){ printf '\n==== %s ====\n' "$*"; }
stop(){ echo "STOP: $*" >&2; exit 1; }

command -v gcloud >/dev/null 2>&1 || stop "gcloud is required"
command -v curl >/dev/null 2>&1 || stop "curl is required"
command -v jq >/dev/null 2>&1 || stop "jq is required"

say "Project identity"
ACTIVE_PROJECT_BEFORE="$(gcloud config get-value project 2>/dev/null || true)"
echo "Cloud Shell project before switch: ${ACTIVE_PROJECT_BEFORE:-<none>}"
echo "Switching Cloud Shell to required FIDUNIO project: $PROJECT"
gcloud config set project "$PROJECT" >/dev/null
ACTIVE_PROJECT="$(gcloud config get-value project 2>/dev/null)"
[[ "$ACTIVE_PROJECT" == "$PROJECT" ]] || stop "Project switch failed; active project is '$ACTIVE_PROJECT', expected '$PROJECT'."
PROJECT_NUMBER="$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')"
[[ "$PROJECT_NUMBER" == "$PROJECT_NUMBER_EXPECTED" ]] || stop "Unexpected project number '$PROJECT_NUMBER'."
echo "Project: $PROJECT ($PROJECT_NUMBER)"
echo "Pinned source: $COMMIT"
echo "Expected runtime service account: $SERVICE_ACCOUNT"

gcloud iam service-accounts describe "$SERVICE_ACCOUNT" --project="$PROJECT" >/dev/null 2>&1 ||
  stop "Dedicated message-delete service account does not exist."

say "Current account-deletion Functions"
present=0
healthy=0
for fn in "${FUNCTIONS[@]}"; do
  if gcloud functions describe "$fn" --gen2 --project="$PROJECT" --region="$REGION" --format=json >"/tmp/${fn}.json" 2>/dev/null; then
    present=$((present+1))
    state="$(jq -r '.state // "MISSING"' "/tmp/${fn}.json")"
    runtime_sa="$(jq -r '.serviceConfig.serviceAccountEmail // "MISSING"' "/tmp/${fn}.json")"
    updated="$(jq -r '.updateTime // ""' "/tmp/${fn}.json")"
    printf '%s\tstate=%s\truntime=%s\tupdated=%s\n' "$fn" "$state" "$runtime_sa" "$updated"
    if [[ "$state" == "ACTIVE" && "$runtime_sa" == "$SERVICE_ACCOUNT" ]]; then
      healthy=$((healthy+1))
    fi
  else
    printf '%s\tABSENT\n' "$fn"
  fi
done

say "Current project-level roles visible for deletion runtime identity"
POLICY="$(mktemp)"
trap 'rm -f "$POLICY"; [[ -z "${WORK:-}" ]] || rm -rf "$WORK"' EXIT
gcloud projects get-iam-policy "$PROJECT" --format=json > "$POLICY"
roles="$(jq -r --arg member "serviceAccount:$SERVICE_ACCOUNT" '
  .bindings[]? | select(any(.members[]?; . == $member)) | .role
' "$POLICY" | sort -u)"
if [[ -n "$roles" ]]; then
  printf '%s\n' "$roles"
else
  echo "No project-level role binding was returned for $SERVICE_ACCOUNT."
fi
echo
echo "NOTE: This is evidence only. This script will NOT add or change IAM."
echo "The final deletion adapter needs Firestore read/write/delete authority and Firebase Auth user-deletion authority."

if [[ "$present" -eq "${#FUNCTIONS[@]}" && "$healthy" -eq "${#FUNCTIONS[@]}" ]]; then
  say "Result"
  echo "PASS: all five account-deletion callables are already ACTIVE under the expected runtime service account."
  echo "NO DEPLOYMENT WAS PERFORMED."
  echo "If the device still reports 'Missing or insufficient permissions', reproduce once and record whether it occurs:"
  echo "  1) immediately after tapping Delete My Account,"
  echo "  2) while removing sent messages,"
  echo "  3) while leaving a group, or"
  echo "  4) during final deletion."
  exit 0
fi

if [[ "$present" -ne 0 ]]; then
  say "Result"
  echo "PARTIAL OR MISALIGNED LIVE STATE: $present/${#FUNCTIONS[@]} functions exist; $healthy are ACTIVE with the expected runtime identity."
  echo "NO DEPLOYMENT WAS PERFORMED."
  stop "Refusing a blind overwrite. Review the function rows above before any live repair."
fi

say "Result"
echo "All five account-deletion callables are ABSENT."
if [[ "$APPROVE_DEPLOY" != "YES" ]]; then
  echo "READ-ONLY PRECHECK COMPLETE. NO DEPLOYMENT WAS PERFORMED."
  echo "After explicit approval, rerun exactly:"
  echo "  APPROVE_DEPLOY=YES bash ad.txt"
  exit 2
fi

say "Pinned source preflight"
if ! command -v firebase >/dev/null 2>&1; then
  echo "Firebase CLI not found; installing firebase-tools..."
  npm install -g firebase-tools
fi
WORK="$(mktemp -d)"
ARCHIVE="$WORK/fidunio.tar.gz"
ROOT="$WORK/hermes-$COMMIT"
curl --retry 3 --retry-delay 2 -fsSL "https://github.com/willyros01/hermes/archive/$COMMIT.tar.gz" -o "$ARCHIVE"
tar -xzf "$ARCHIVE" -C "$WORK"

for marker in   'export const getMyAccountDeletionRequestV1=onCall'   'export const requestMyAccountDeletionV1=onCall'   'export const markMyAccountDeletionCleanupCompleteV1=onCall'   'export const cancelMyAccountDeletionRequestV1=onCall'   'export const completeMyAccountDeletionV1=onCall'
do
  grep -Fq "$marker" "$ROOT/functions/index.mjs" || stop "Pinned source is missing expected export: $marker"
done
grep -Fq 'const accountDeletionRepo=createAccountDeletionAdminRepository({db,auth:getAuth()});' "$ROOT/functions/index.mjs" ||
  stop "Pinned source is missing the reviewed account-deletion Admin repository."
grep -Fq 'async deleteAuthUser(uid){await auth.deleteUser(uid);}' "$ROOT/functions/account-deletion/account-deletion-firestore-admin-adapter.mjs" ||
  stop "Pinned source is missing Firebase Auth deletion-last implementation."

npm --prefix "$ROOT/functions" install --ignore-scripts --silent
(
  cd "$ROOT"
  node --input-type=module <<'JS'
const m=await import('./functions/index.mjs');
for(const name of [
  'getMyAccountDeletionRequestV1',
  'requestMyAccountDeletionV1',
  'markMyAccountDeletionCleanupCompleteV1',
  'cancelMyAccountDeletionRequestV1',
  'completeMyAccountDeletionV1'
]){
  if(typeof m[name] !== 'function') throw new Error(name+' export missing');
}
console.log('PASS: pinned account-deletion Functions graph imports successfully');
JS
)

say "Controlled deployment"
cd "$ROOT"
firebase deploy \
  --project "$PROJECT" \
  --only "functions:$CODEBASE:getMyAccountDeletionRequestV1,functions:$CODEBASE:requestMyAccountDeletionV1,functions:$CODEBASE:markMyAccountDeletionCleanupCompleteV1,functions:$CODEBASE:cancelMyAccountDeletionRequestV1,functions:$CODEBASE:completeMyAccountDeletionV1" \
  --non-interactive

say "Post-deploy verification"
for fn in "${FUNCTIONS[@]}"; do
  state="$(gcloud functions describe "$fn" --gen2 --project="$PROJECT" --region="$REGION" --format='value(state)')"
  runtime_sa="$(gcloud functions describe "$fn" --gen2 --project="$PROJECT" --region="$REGION" --format='value(serviceConfig.serviceAccountEmail)')"
  [[ "$state" == "ACTIVE" ]] || stop "$fn deployment state is '$state'."
  [[ "$runtime_sa" == "$SERVICE_ACCOUNT" ]] || stop "$fn runtime service account is '$runtime_sa', expected '$SERVICE_ACCOUNT'."
  echo "PASS: $fn ACTIVE -> $runtime_sa"
done

cat <<'ROLLBACK'

DEPLOYMENT COMPLETE.

Rollback boundary:
- This script deploys only when all five callables were absent before the change.
- Therefore rollback to the captured pre-change state is removal of only these five newly-added callables.
- Do NOT run rollback unless the deployment is rejected and rollback is explicitly approved.

Rollback command:
  firebase functions:delete \
    getMyAccountDeletionRequestV1 \
    requestMyAccountDeletionV1 \
    markMyAccountDeletionCleanupCompleteV1 \
    cancelMyAccountDeletionRequestV1 \
    completeMyAccountDeletionV1 \
    --region us-central1 \
    --project fidunio-fef13 \
    --force

No Firestore rules, IAM bindings, App Check configuration, Storage rules, or application data are changed by this handoff.
ROLLBACK
