#!/usr/bin/env bash
# FIDUNIO Android: create ONE permanent APK key and copy-ready secret text files.
# Run in YOUR Google Cloud Shell. Do not commit, upload or paste keys into chat.
set -euo pipefail
umask 077
gcloud config set project fidunio-fef13

# A previous signing identity must NEVER be replaced automatically.
if [ -e "$HOME/fidunio.p12" ]; then
  echo "Existing fidunio.p12 found in home. STOP: do not create a second signing key."
  exit 1
fi
DIR="$HOME/fidunio-sign"
mkdir -p "$DIR"
cd "$DIR"
if [ -e fidunio.p12 ] || [ -e fkey.txt ] || [ -e fpas.txt ]; then
  echo "FIDUNIO signing files already exist. STOP: preserve your original permanent key."
  echo "Location: $DIR"
  exit 1
fi

# Generate a strong unique password without manual typing.
openssl rand -hex 24 > fpas.txt
P="$(cat fpas.txt)"
keytool -genkeypair -storetype PKCS12 -keystore fidunio.p12 -alias fidunio \
  -keyalg RSA -keysize 3072 -validity 10000 \
  -dname 'CN=FIDUNIO Android, OU=Cuberoot, O=Cuberoot Systems' \
  -storepass "$P" -keypass "$P" -noprompt
keytool -list -keystore fidunio.p12 -storetype PKCS12 \
  -storepass "$P" -alias fidunio > /dev/null
base64 -w0 fidunio.p12 > fkey.txt
printf '\n' >> fkey.txt
chmod 600 fidunio.p12 fkey.txt fpas.txt
unset P

echo
echo "SUCCESS: permanent FIDUNIO Android signing key created."
echo "Secret 1 name: FIDUNIO_ANDROID_KEYSTORE_BASE64"
echo "Secret 1 value: entire contents of $DIR/fkey.txt"
echo "Secret 2 name: FIDUNIO_ANDROID_PASSWORD"
echo "Secret 2 value: entire contents of $DIR/fpas.txt"
echo
echo "Use: cat $DIR/fkey.txt   (copy everything shown for Secret 1)"
echo "Use: cat $DIR/fpas.txt   (copy everything shown for Secret 2)"
echo "PRIVATE BACKUP REQUIRED: $DIR/fidunio.p12 plus fpas.txt"
echo "No GitHub secrets were uploaded. No APK build was started by this script."
