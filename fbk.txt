#!/usr/bin/env bash
# FIDUNIO — NIGHTLY BACKUP SETUP, run ONCE (prepared by the assistant, Oct 2, 2026)
#
# Run:     bash fbk.txt
# Initial backup and access setup; no optional parameters.
#
# Creates the READ-ONLY helper the nightly backup on GitHub signs in as:
#   1. a helper account, fidunio-backup, that can only READ the database and
#      the list of sign-in accounts — it cannot change or delete anything;
#   2. a keyless sign-in for GitHub that only the nightly-backup workflow of the
#      willyros01/hermes repository, on main, can use. No key file exists.
# Reads the database for an initial backup, then sets up read-only nightly access.

# ---------------------------------------------------------------------------
# THE GOOGLE PROJECT — defined first, before any command runs.
# ---------------------------------------------------------------------------
PROJECT_ID="fidunio-fef13"
unset CLOUDSDK_CORE_PROJECT GOOGLE_CLOUD_PROJECT

set -Eeuo pipefail

echo "Switching Cloud Shell to project ${PROJECT_ID} ..."
gcloud config set project "${PROJECT_ID}" --quiet >/dev/null 2>&1 || { echo "ERROR: could not switch to project ${PROJECT_ID}. Nothing changed." >&2; exit 1; }
CURRENT_PROJECT="$(gcloud config get-value project 2>/dev/null)"
[[ "${CURRENT_PROJECT}" == "${PROJECT_ID}" ]] || { echo "ERROR: Cloud Shell is on '${CURRENT_PROJECT}', not ${PROJECT_ID}. Nothing changed." >&2; exit 1; }
echo "OK: Cloud Shell is now on project ${CURRENT_PROJECT}."
export CLOUDSDK_CORE_PROJECT="${PROJECT_ID}"
export GOOGLE_CLOUD_PROJECT="${PROJECT_ID}"

NUMBER="130339622893"                     # the project number the workflow uses
SA="fidunio-backup@${PROJECT_ID}.iam.gserviceaccount.com"
POOL="fidunio-backup"
PROVIDER="nightly-backup"
REPO_ID="1354087461"                      # GitHub's id for willyros01/hermes (survives a rename)
WORKFLOW="willyros01/hermes/.github/workflows/hbk.yml@refs/heads/main"
CONDITION="assertion.repository_id=='${REPO_ID}' && assertion.workflow_ref=='${WORKFLOW}'"
MAPPING="google.subject=assertion.sub,attribute.repository_id=assertion.repository_id,attribute.workflow_ref=assertion.workflow_ref"
MEMBER="principalSet://iam.googleapis.com/projects/${NUMBER}/locations/global/workloadIdentityPools/${POOL}/*"
ROLES=(roles/datastore.viewer roles/firebaseauth.viewer)

die(){ echo "ERROR: $*" >&2; exit 1; }
command -v gcloud >/dev/null || die "gcloud is unavailable. Run this in Google Cloud Shell."
ACCOUNT="$(gcloud auth list --filter=status:ACTIVE --format='value(account)' | head -n1)"
[[ -n "${ACCOUNT}" ]] || die "No active Google account. Sign in to Cloud Shell first."
FOUND="$(gcloud projects describe "${PROJECT_ID}" --format='value(projectNumber)' 2>/dev/null || true)"
[[ "${FOUND}" == "${NUMBER}" ]] || die "Project ${PROJECT_ID} has number '${FOUND}', expected ${NUMBER}. Nothing changed. Send a screenshot to the assistant."
echo
echo "  Google account : ${ACCOUNT}"
echo "  Project        : ${PROJECT_ID} (${NUMBER})"
echo

retry(){ local n; for n in 1 2 3 4 5 6; do "$@" && return 0; sleep $((n * ${NBK_SLEEP:-5})); done; return 1; }
has_role(){ gcloud projects get-iam-policy "${PROJECT_ID}" --flatten='bindings[].members' \
  --filter="bindings.role:$1 AND bindings.members:${SA}" --format='value(bindings.role)' | grep -qx "$1"; }
sa_ok(){ gcloud iam service-accounts describe "${SA}" --format='value(email)' >/dev/null 2>&1; }
pool_state(){ gcloud iam workload-identity-pools describe "${POOL}" --location=global --format='value(state)' 2>/dev/null || true; }
provider_ok(){
  local c; c="$(gcloud iam workload-identity-pools providers describe "${PROVIDER}" --location=global \
    --workload-identity-pool="${POOL}" --format='value(attributeCondition)' 2>/dev/null)" || return 1
  [[ "${c}" == "${CONDITION}" ]]
}
link_ok(){ gcloud iam service-accounts get-iam-policy "${SA}" --flatten='bindings[].members' \
  --filter="bindings.role:roles/iam.workloadIdentityUser" --format='value(bindings.members)' | grep -Fqx "${MEMBER}"; }

verify(){
  local bad=0
  sa_ok && echo "OK: the helper account exists." || { echo "MISSING: the helper account."; bad=1; }
  for r in "${ROLES[@]}"; do has_role "$r" && echo "OK: it can read ($r)." || { echo "MISSING: $r"; bad=1; }; done
  [[ "$(pool_state)" == ACTIVE ]] && echo "OK: the keyless sign-in is set up." || { echo "MISSING: the keyless sign-in."; bad=1; }
  provider_ok && echo "OK: only the nightly-backup workflow on main can use it." || { echo "MISSING: the GitHub link."; bad=1; }
  link_ok && echo "OK: the workflow may act as the helper." || { echo "MISSING: the helper link."; bad=1; }
  return "${bad}"
}

echo "This creates a READ-ONLY helper for the nightly backup. No data is changed."



# Make the initial encrypted backup BEFORE any setup change.
WORK="$(mktemp -d)"
trap 'rm -rf "${WORK}"' EXIT
DEST="${HOME}/fidunio-backups"
mkdir -p "${DEST}"
chmod 700 "${DEST}"
STAMP="$(date -u +%Y-%m-%d)"
gcloud projects get-iam-policy "${PROJECT_ID}" --format=json > "${DEST}/iam-before-${STAMP}.json"
if sa_ok; then
  gcloud iam service-accounts get-iam-policy "${SA}" --format=json > "${DEST}/sa-before-${STAMP}.json"
fi
gcloud iam workload-identity-pools providers describe "${PROVIDER}" --location=global --workload-identity-pool="${POOL}" --format=json > "${DEST}/wif-before-${STAMP}.json" 2>/dev/null || rm -f "${DEST}/wif-before-${STAMP}.json"
curl -fsSL "https://raw.githubusercontent.com/willyros01/hermes/main/.github/scripts/fbk.mjs" -o "${WORK}/fbk.mjs"
read -r -s -p "Enter your backup passphrase (16 characters or more; hidden): " BACKUP_PASS
echo
[[ ${#BACKUP_PASS} -ge 16 ]] || die "The passphrase must have 16 characters or more. Nothing changed."
read -r -s -p "Enter it again (hidden): " BACKUP_CONFIRM
echo
[[ "${BACKUP_PASS}" == "${BACKUP_CONFIRM}" ]] || die "Passphrases do not match. Nothing changed."
unset BACKUP_CONFIRM
ACCESS_TOKEN="$(gcloud auth print-access-token)" PROJECT="${PROJECT_ID}" node "${WORK}/fbk.mjs" "${WORK}/backup.json"
jq -e '.kind == "fidunio-full-backup" and .project == "fidunio-fef13" and (.accounts|type == "array") and ([.documents[]|select((.missing|not) and (.path|test("^users/[^/]+$")))]|length > 0)' "${WORK}/backup.json" >/dev/null || die "Backup is incomplete. Nothing changed."
export GNUPGHOME="${WORK}/gnupg"
mkdir -m 700 "${GNUPGHOME}"
INITIAL="${DEST}/fidunio-${STAMP}.json.gpg"
gpg --batch --quiet --yes --no-tty --pinentry-mode loopback --passphrase-fd 3 --symmetric --cipher-algo AES256 --s2k-mode 3 --s2k-digest-algo SHA512 --s2k-count 65011712 --compress-algo zlib --output "${WORK}/backup.gpg" "${WORK}/backup.json" 3<<<"${BACKUP_PASS}"
gpg --batch --quiet --yes --no-tty --pinentry-mode loopback --passphrase-fd 3 --decrypt --output "${WORK}/check.json" "${WORK}/backup.gpg" 3<<<"${BACKUP_PASS}"
cmp -s "${WORK}/backup.json" "${WORK}/check.json" || die "Encrypted backup verification failed. Nothing changed."
cp "${WORK}/backup.gpg" "${INITIAL}"
chmod 600 "${INITIAL}"
sha256sum "${INITIAL}" > "${INITIAL}.sha256"
echo "INITIAL DATABASE BACKUP VERIFIED: ${INITIAL}"
jq -r '"Documents: \(.documents|length). Sign-in accounts: \(.accounts|length). Saved: \(.savedAt)."' "${WORK}/backup.json"
unset BACKUP_PASS
rm -f "${WORK}/backup.json" "${WORK}/check.json"
# Rollback for resources created by this script only; existing resources stay intact.
ROLLBACK="${DEST}/rbk.txt"
printf '%s\n' '#!/usr/bin/env bash' 'set -euo pipefail' 'PROJECT_ID="fidunio-fef13"' > "${ROLLBACK}"
if ! sa_ok; then
  printf '%s\n' 'gcloud iam service-accounts delete fidunio-backup@fidunio-fef13.iam.gserviceaccount.com --project="$PROJECT_ID" --quiet' >> "${ROLLBACK}"
else
  for r in "${ROLES[@]}"; do
    if ! has_role "${r}"; then
      printf '%s\n' "gcloud projects remove-iam-policy-binding fidunio-fef13 --member=serviceAccount:fidunio-backup@fidunio-fef13.iam.gserviceaccount.com --role=${r} --condition=None --quiet" >> "${ROLLBACK}"
    fi
  done
fi
if [[ -z "$(pool_state)" ]]; then
  printf '%s\n' 'gcloud iam workload-identity-pools delete fidunio-backup --project="$PROJECT_ID" --location=global --quiet' >> "${ROLLBACK}"
elif [[ ! -f "${DEST}/wif-before-${STAMP}.json" ]]; then
  printf '%s\n' 'gcloud iam workload-identity-pools providers delete nightly-backup --project="$PROJECT_ID" --location=global --workload-identity-pool=fidunio-backup --quiet' >> "${ROLLBACK}"
else
  provider_ok || die "An existing provider has different trust settings. Backup is saved; IAM setup stopped for review."
fi
if sa_ok && ! link_ok; then
  printf '%s\n' "gcloud iam service-accounts remove-iam-policy-binding fidunio-backup@fidunio-fef13.iam.gserviceaccount.com --project=fidunio-fef13 --role=roles/iam.workloadIdentityUser --member=${MEMBER} --quiet" >> "${ROLLBACK}"
fi
echo "1. Switching on the sign-in services (free) ..."
gcloud services enable iam.googleapis.com iamcredentials.googleapis.com sts.googleapis.com --project="${PROJECT_ID}" --quiet >/dev/null \
  || die "Could not switch on the sign-in services. Send a screenshot to the assistant."

echo "2. The read-only helper account ..."
if ! sa_ok; then
  gcloud iam service-accounts create fidunio-backup --display-name="FIDUNIO nightly backup (read only)" --quiet >/dev/null \
    || die "Could not create the helper account. Send a screenshot to the assistant."
fi
for r in "${ROLES[@]}"; do
  has_role "$r" || retry gcloud projects add-iam-policy-binding "${PROJECT_ID}" --member="serviceAccount:${SA}" \
    --role="$r" --condition=None --quiet >/dev/null || die "Could not give the helper ${r}. Send a screenshot to the assistant."
done

echo "3. The keyless sign-in for GitHub ..."
case "$(pool_state)" in
  ACTIVE) ;;
  DELETED) gcloud iam workload-identity-pools undelete "${POOL}" --location=global --quiet >/dev/null || die "Could not restore the sign-in pool." ;;
  *) gcloud iam workload-identity-pools create "${POOL}" --location=global --display-name="FIDUNIO nightly backup" --quiet >/dev/null \
       || die "Could not create the sign-in pool. Send a screenshot to the assistant." ;;
esac
if gcloud iam workload-identity-pools providers describe "${PROVIDER}" --location=global --workload-identity-pool="${POOL}" >/dev/null 2>&1; then
  gcloud iam workload-identity-pools providers update-oidc "${PROVIDER}" --location=global --workload-identity-pool="${POOL}" \
    --attribute-mapping="${MAPPING}" --attribute-condition="${CONDITION}" --quiet >/dev/null || die "Could not update the GitHub link."
else
  retry gcloud iam workload-identity-pools providers create-oidc "${PROVIDER}" --location=global --workload-identity-pool="${POOL}" \
    --display-name="GitHub nightly backup" --issuer-uri="https://token.actions.githubusercontent.com" \
    --attribute-mapping="${MAPPING}" --attribute-condition="${CONDITION}" --quiet >/dev/null || die "Could not create the GitHub link. Send a screenshot to the assistant."
fi

echo "4. Letting that workflow act as the helper ..."
link_ok || retry gcloud iam service-accounts add-iam-policy-binding "${SA}" --role=roles/iam.workloadIdentityUser \
  --member="${MEMBER}" --quiet >/dev/null || die "Could not link the workflow to the helper. Send a screenshot to the assistant."

echo
echo "5. Checking ..."
verify || die "Something is missing (see above). Run bash fbk.txt again; if it repeats, send a screenshot to the assistant."
echo
echo "DONE. Initial encrypted backup: ${INITIAL}"
echo "Rollback of new backup-access setup only: bash ${ROLLBACK}"
echo "Use the SAME passphrase in Hermes GitHub Actions secret BACKUP_PASSPHRASE."
echo "Tell the assistant the final result so the first GitHub nightly run can be verified."
