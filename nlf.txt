#!/usr/bin/env bash
# FIDUNIO 1.1.76 notification lease Cloud Functions deployment
#
# RUN WITH:   bash nlf.txt
#
# WHAT IT DEPLOYS (two Cloud Functions ONLY)
# - claimNotificationLeaseV1:  leases now last 30 days (was 24 hours), so a phone keeps
#                               getting notifications when FIDUNIO is not opened for a while.
# - revokeNotificationLeaseV1: sign-out keeps a revoked marker so a stale registration of
#                               another account on the same phone cannot start receiving again.
# These functions serve both the iOS app and the web app.
#
# SAFETY
# - Switches to project fidunio-fef13 first and verifies project number 130339622893.
# - Downloads the current ios commit and refuses to continue unless every deployed source
#   file is byte-for-byte the reviewed release version (SHA-256 below).
# - Runs the notification lease regression suites before deploying.
# - Writes a rollback script (redeploys the previous version, commit 68fb57d) before deploying.
# - Verifies both functions are ACTIVE under the dedicated notification service account.
# - Does NOT touch any other function, Firestore/Storage rules, IAM, App Check or data.
set -euo pipefail

PROJECT="fidunio-fef13"
PROJECT_NUMBER_EXPECTED="130339622893"
REPO="willyros01/hermes"
BRANCH="ios"
REGION="us-central1"
CODEBASE="recovery"
NOTIFICATION_SA="fidunio-notification@fidunio-fef13.iam.gserviceaccount.com"
FUNCTIONS=(claimNotificationLeaseV1 revokeNotificationLeaseV1)
ROLLBACK_COMMIT="68fb57d01b9b716e18f2a1c8943466f472064e70"
declare -A EXPECTED_SHA256=(
  [functions/index.mjs]="6e898034ca09c1b31b500d30f9706d50726717169994d4a6ba0518e64e27a558"
  [functions/notification/notification-lease-policy.mjs]="622fd69f3a701b39da734f1d9e296527d31a0fff7dae540b91ca7b53093c232f"
  [functions/notification/notification-lease-firestore-admin.mjs]="205a7f4782baf9e9834e16346ba133ed09717d2f4e1c38a5c59120d3bcda778f"
  [functions/package.json]="b92fa95d6eb27e30c79a221459776c4aa601a5772142180815b3f611cda6102f"
  [functions/package-lock.json]="f543212e3d79013d3a98744d3fcf099867db86435cfd6c235ad7ee5dff900a36"
)

say(){ printf '\n==== %s ====\n' "$*"; }
stop(){ echo "STOP: $*" >&2; echo "NOTHING WAS DEPLOYED unless a PASS line above says so." >&2; exit 1; }
for cmd in gcloud curl jq npm node tar sha256sum; do
  command -v "$cmd" >/dev/null 2>&1 || stop "$cmd is required"
done

say "Select and verify the FIDUNIO project"
gcloud config set project "$PROJECT" >/dev/null 2>&1 || stop "could not select $PROJECT"
ACTIVE="$(gcloud config get-value project 2>/dev/null)"
[[ "$ACTIVE" == "$PROJECT" ]] || stop "active project is '$ACTIVE'"
NUMBER="$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')"
[[ "$NUMBER" == "$PROJECT_NUMBER_EXPECTED" ]] || stop "wrong project number '$NUMBER'"
gcloud iam service-accounts describe "$NOTIFICATION_SA" --project="$PROJECT" >/dev/null 2>&1 || stop "notification service account is missing"
echo "PASS: $PROJECT ($NUMBER), service account present"

STAMP="$(date -u +%Y%m%dT%H%M%SZ)"
BACKUP_DIR="$HOME/fidunio-lease-functions-1.1.76-$STAMP"
WORK="$(mktemp -d)"
mkdir -p "$BACKUP_DIR"
trap 'rm -rf "$WORK"' EXIT

fetch(){ # fetch <commit> <dir>
  curl --retry 3 --retry-delay 2 -fsSL "https://github.com/$REPO/archive/$1.tar.gz" -o "$WORK/$1.tar.gz"
  tar -xzf "$WORK/$1.tar.gz" -C "$WORK"
  [[ -d "$WORK/hermes-$1" ]] || stop "archive for $1 did not unpack"
}

say "Download the current ios commit and verify the reviewed source"
COMMIT="$(curl --retry 3 -fsSL "https://api.github.com/repos/$REPO/commits/$BRANCH" | jq -r '.sha // empty')"
[[ "$COMMIT" =~ ^[0-9a-f]{40}$ ]] || stop "could not resolve the $BRANCH commit"
echo "Source commit: $COMMIT"
fetch "$COMMIT"
ROOT="$WORK/hermes-$COMMIT"
for f in "${!EXPECTED_SHA256[@]}"; do
  [[ -f "$ROOT/$f" ]] || stop "missing $f"
  got="$(sha256sum "$ROOT/$f" | awk '{print $1}')"
  [[ "$got" == "${EXPECTED_SHA256[$f]}" ]] || stop "$f is not the reviewed release version. Ask Claude for an updated script."
done
grep -Fq 'export const NOTIFICATION_LEASE_MS = 30*24*60*60*1000;' "$ROOT/functions/notification/notification-lease-policy.mjs" || stop "lease length is not 30 days"
grep -Fq 'if(matching)tx.set(ownerRef,{...owner,enabled:false,leaseUntilMs:0,revokedMs:now()},{merge:false});' "$ROOT/functions/notification/notification-lease-firestore-admin.mjs" || stop "revoke tombstone missing"
echo "PASS: reviewed release source"

say "Run the notification lease regression suites"
(
  cd "$ROOT"
  node notification-token-lease.test.mjs
  node functions/notification/notification-lease-policy.test.mjs
  node functions/notification/notification-lease-firestore-admin.test.mjs
  node functions/notification/direct-message-notification-core.test.mjs
  node functions/notification/group-message-notification-core.test.mjs
) || stop "a notification lease test failed"
echo "PASS: notification lease regression suites"

say "Install and import-check the Functions source"
command -v firebase >/dev/null 2>&1 || npm install -g firebase-tools >/dev/null
npm --prefix "$ROOT/functions" ci --ignore-scripts --no-audit --no-fund >/dev/null
( cd "$ROOT" && node --input-type=module -e "const m=await import('./functions/index.mjs');for(const n of ['claimNotificationLeaseV1','revokeNotificationLeaseV1'])if(typeof m[n]!=='function')throw new Error('missing '+n);console.log('PASS: Functions source imports and exports both lease functions')" )

say "Record current state and prepare rollback (previous version: $ROLLBACK_COMMIT)"
for fn in "${FUNCTIONS[@]}"; do
  gcloud functions describe "$fn" --gen2 --project="$PROJECT" --region="$REGION" --format=json > "$BACKUP_DIR/$fn-before.json" 2>/dev/null || echo "{}" > "$BACKUP_DIR/$fn-before.json"
  echo "$fn before: state=$(jq -r '.state // "NOT DEPLOYED"' "$BACKUP_DIR/$fn-before.json") updated=$(jq -r '.updateTime // "-"' "$BACKUP_DIR/$fn-before.json")"
done
ONLY="$(printf "functions:$CODEBASE:%s," "${FUNCTIONS[@]}")"; ONLY="${ONLY%,}"
cat > "$BACKUP_DIR/rollback.sh" <<ROLLBACK
#!/usr/bin/env bash
set -euo pipefail
gcloud config set project "$PROJECT" >/dev/null
[[ "\$(gcloud config get-value project 2>/dev/null)" == "$PROJECT" ]] || { echo "STOP: wrong project"; exit 1; }
[[ "\$(gcloud projects describe $PROJECT --format='value(projectNumber)')" == "$PROJECT_NUMBER_EXPECTED" ]] || { echo "STOP: wrong project number"; exit 1; }
W="\$(mktemp -d)"; trap 'rm -rf "\$W"' EXIT
curl --retry 3 -fsSL "https://github.com/$REPO/archive/$ROLLBACK_COMMIT.tar.gz" | tar -xz -C "\$W"
cd "\$W/hermes-$ROLLBACK_COMMIT"
npm --prefix functions ci --ignore-scripts --no-audit --no-fund >/dev/null
firebase deploy --project "$PROJECT" --only "$ONLY" --non-interactive
echo "ROLLBACK COMPLETE: previous notification lease functions restored."
ROLLBACK
chmod +x "$BACKUP_DIR/rollback.sh"
echo "Rollback ready: bash $BACKUP_DIR/rollback.sh"

say "Re-check project, then deploy the two lease functions ONLY"
gcloud config set project "$PROJECT" >/dev/null
[[ "$(gcloud config get-value project 2>/dev/null)" == "$PROJECT" ]] || stop "project changed"
[[ "$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')" == "$PROJECT_NUMBER_EXPECTED" ]] || stop "project number changed"
( cd "$ROOT" && firebase deploy --project "$PROJECT" --only "$ONLY" --non-interactive )

say "Verify both functions are ACTIVE under the notification service account"
for fn in "${FUNCTIONS[@]}"; do
  gcloud functions describe "$fn" --gen2 --project="$PROJECT" --region="$REGION" --format=json > "$BACKUP_DIR/$fn-after.json"
  state="$(jq -r '.state // empty' "$BACKUP_DIR/$fn-after.json")"
  sa="$(jq -r '.serviceConfig.serviceAccountEmail // empty' "$BACKUP_DIR/$fn-after.json")"
  before="$(jq -r '.updateTime // ""' "$BACKUP_DIR/$fn-before.json")"
  after="$(jq -r '.updateTime // ""' "$BACKUP_DIR/$fn-after.json")"
  [[ "$state" == "ACTIVE" ]] || stop "$fn state is '$state'. Roll back with: bash $BACKUP_DIR/rollback.sh"
  [[ "$sa" == "$NOTIFICATION_SA" ]] || stop "$fn runs as '$sa'. Roll back with: bash $BACKUP_DIR/rollback.sh"
  [[ "$after" != "$before" ]] || stop "$fn was not updated (update time unchanged)"
  echo "PASS: $fn ACTIVE as $sa (updated $after)"
done

say "SUCCESS"
echo "FIDUNIO 1.1.76 notification lease functions are live (source commit $COMMIT)."
echo "Rollback: bash $BACKUP_DIR/rollback.sh"
echo "No other function, rule, IAM or App Check setting was changed."
