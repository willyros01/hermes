#!/usr/bin/env bash
set -euo pipefail

PROJECT="fidunio-fef13"
COMMIT="5a6449c9165228a0f7cf75418f3dbecadc736f99"
SCRIPT_VERSION="1.1.36-a"

ACTIVE_PROJECT="$(gcloud config get-value project 2>/dev/null)"
if [[ "$ACTIVE_PROJECT" != "$PROJECT" ]]; then
  echo "STOP: Cloud Shell project is '$ACTIVE_PROJECT', expected '$PROJECT'."
  exit 1
fi
if ! command -v firebase >/dev/null 2>&1; then
  echo "Firebase CLI not found; installing firebase-tools..."
  npm install -g firebase-tools
fi

WORK="$(mktemp -d)"
trap 'rm -rf "$WORK"' EXIT
ARCHIVE="$WORK/fidunio.tar.gz"
ROOT="$WORK/hermes-$COMMIT"

echo "FIDUNIO 1.1.36 group history-boundary rules deployment"
echo "Project: $PROJECT"
echo "Pinned source: $COMMIT"
echo "Script: $SCRIPT_VERSION"

curl --retry 3 --retry-delay 2 -fsSL "https://github.com/willyros01/hermes/archive/$COMMIT.tar.gz" -o "$ARCHIVE"
tar -xzf "$ARCHIVE" -C "$WORK"

if ! grep -Fq 'version: "1.1.36"' "$ROOT/version.js"; then
  echo "STOP: pinned source is not FIDUNIO 1.1.36."
  exit 1
fi
if ! grep -Fq 'readCloudGroupMessageAccess' "$ROOT/firebase.js" || ! grep -Fq 'where("createdAt",">=",access.historyFrom)' "$ROOT/firebase.js"; then
  echo "STOP: pinned source does not contain the reviewed join-time query boundary."
  exit 1
fi
if ! grep -Fq 'function groupMessageReadable' "$ROOT/firestore.rules" || ! grep -Fq 'request.auth.uid in resource.data.envelopes' "$ROOT/firestore.rules"; then
  echo "STOP: pinned Firestore rules do not contain the reviewed message and epoch boundaries."
  exit 1
fi

cd "$ROOT"
firebase deploy \
  --project "$PROJECT" \
  --only "firestore:rules" \
  --non-interactive

ACTIVE_RULESET="$(python3 - "$PROJECT" <<'PY'
import json
import subprocess
import sys
import urllib.request

project = sys.argv[1]
token = subprocess.check_output(["gcloud", "auth", "print-access-token"], text=True).strip()
headers = {"Authorization": f"Bearer {token}"}

def read_json(url):
    request = urllib.request.Request(url, headers=headers)
    with urllib.request.urlopen(request) as response:
        return json.load(response)

root = f"https://firebaserules.googleapis.com/v1/projects/{project}"
release = read_json(f"{root}/releases/cloud.firestore")
ruleset_name = release["rulesetName"]
ruleset = read_json(f"https://firebaserules.googleapis.com/v1/{ruleset_name}")
source = "\n".join(file.get("content", "") for file in ruleset.get("source", {}).get("files", []))
required = (
    "function groupMessageReadable",
    "request.auth.uid in resource.data.envelopes",
)
if not all(marker in source for marker in required):
    raise SystemExit("STOP: active Firestore rules do not contain the reviewed boundaries.")
print(ruleset_name)
PY
)"

echo "SUCCESS"
echo "FIDUNIO 1.1.36 Firestore group history-boundary rules are deployed."
echo "Pinned source: $COMMIT"
echo "Active ruleset: $ACTIVE_RULESET"
echo "No Functions, indexes, Storage rules, or application data were changed."
